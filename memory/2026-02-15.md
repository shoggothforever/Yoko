### Nginx 服务器配置

#### 配置文件内容
```nginx
server {
    listen 80;
    listen [::]:80;
    server_name 118.145.99.224;
    
    # 重定向到 HTTPS
    return 301 https://$server_name$request_uri;
}

server {
    listen 443 ssl http2;
    listen [::]:443 ssl http2;
    server_name 118.145.99.224;
    
    root /root/.openclaw/workspace/yoko-blog;
    index index.html;
    
    # SSL 证书配置
    ssl_certificate /etc/letsencrypt/live/yoko.sfct.top/fullchain.pem;
    ssl_certificate_key /etc/letsencrypt/live/yoko.sfct.top/privkey.pem;
    ssl_trusted_certificate /etc/letsencrypt/live/yoko.sfct.top/chain.pem;
    
    # SSL 配置
    ssl_protocols TLSv1.2 TLSv1.3;
    ssl_ciphers HIGH:!aNULL:!MD5;
    ssl_prefer_server_ciphers off;
    
    # 启用gzip压缩
    gzip on;
    gzip_vary on;
    gzip_min_length 1024;
    gzip_types text/plain text/css text/javascript application/javascript application/xml+rss text/xml;
    
    location / {
        try_files $uri $uri/ =404;
    }
}
```
